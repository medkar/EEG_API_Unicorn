 751              "...et ne rejoint PAS la manche en cours (elle n'a rien à décoder)")
 752          rt._vider_manche()
 753  
 754          # Les compteurs de pannes ont une sortie AUTRE que le terminal : `state()`, lu par la
 755          # console (ou n'importe quel autre client du moteur qui n'a pas les yeux sur les logs).
 756          etat = rt.state()
 757          chk({"refus_cible", "epoques_perdues", "manches_abandonnees"} <= set(etat),
 758              f"les trois compteurs de pannes bruyantes sont exposés dans state() ({sorted(etat)})")
 759          chk(etat["refus_cible"] == rt._refus_cible
 760              and etat["epoques_perdues"] == rt._epoques_perdues
 761              and etat["manches_abandonnees"] == rt._manches_abandonnees,
 762              f"...et reflètent les compteurs RÉELS du runtime, pas une copie figée "
 763              f"(state={etat['refus_cible'], etat['epoques_perdues'], etat['manches_abandonnees']}, "
 764              f"réel={rt._refus_cible, rt._epoques_perdues, rt._manches_abandonnees})")
 765  
 766          # 6. Le contrat du mode.
 767          chk(SPEC.rest.duration_s == 0.0 and SPEC.rest.warmup_s == SSVEP_WARMUP_S,
 768              f"chauffe obligatoire, aucun plancher ({SPEC.rest})")
 769          chk(SPEC.stream == "decoded_p300" and SPEC.status == "moteur",
 770              "le mode publie decoded_p300 et tourne dans le moteur")
 771          chk(SPEC.calibration is not None and SPEC.calibration.kind == "natif"
 772              and SPEC.calibration.runtime_cls is None,
 773              "sa calibration reste NATIVE : le moteur ne la joue pas, l'appli pygame la joue")
 774          chk(all(p.affecte_decodage for p in SPEC.params if p.key != "stream_in"),
 775              "le modèle affecte le décodage ; le flux de marqueurs (juste le NOM écouté), non")
 776          chk(P300_ROUND_TIMEOUT_S > 0.0,
 777              f"un délai d'abandon strictement positif est déclaré ({P300_ROUND_TIMEOUT_S:g} s)")
 778          chk(_MAX_EPOQUES > P300_N_TARGETS,
 779              f"le plafond dur dépasse largement une manche normale ({_MAX_EPOQUES} > "
 780              f"{P300_N_TARGETS})")
 781  
 782          # --- LE test d'alignement ------------------------------------------------
 783          # On fabrique un tampon plat, on y plante un pic d'amplitude unique à un instant CONNU, et
 784          # on envoie un marqueur à cet instant. L'époque extraite doit contenir ce pic exactement à
 785          # l'échantillon `n_pre` — c'est-à-dire à l'onset. Un décalage de 3 échantillons (12 ms) ne
 786          # change RIEN d'autre : l'époque a la bonne taille, le décodeur tourne, les scores sortent.
 787          fs = 250.0
 788          n_pre = int(round(P300_PRE_S * fs))       # 37
 789          n_post = int(round(P300_EPOCH_S * fs))    # 200
 790          t0 = 1000.0
 791          ts = np.arange(t0, t0 + 4.0, 1.0 / fs)
 792          eeg = np.zeros((len(ts), 8))
 793          instant_du_pic = t0 + 2.0
 794          i_pic = int(np.searchsorted(ts, instant_du_pic))
 795          eeg[i_pic, :] = 42.0                      # une valeur qu'aucun calcul ne produit par hasard
 796  
 797          epoque = epoch_from_stream(eeg, ts, instant_du_pic, fs,
 798                                    pre_s=P300_PRE_S, post_s=P300_EPOCH_S)
 799          chk(epoque is not None, "l'époque est extraite")
 800          chk(epoque.shape == (n_pre + n_post, 8),
 801              f"elle a exactement pré+post échantillons ({epoque.shape})")
 802          position = int(np.argmax(epoque[:, 0]))
 803          chk(position == n_pre,
 804              f"⚠️ ALIGNEMENT : le pic planté à l'onset se retrouve à l'échantillon {position}, "
 805              f"il devait être à {n_pre} (décalage de {position - n_pre} échantillons = "
 806              f"{(position - n_pre) / fs * 1000:+.0f} ms)")
 807          chk(abs(epoque[n_pre, 0] - 42.0) < 1e-9,
 808              f"et c'est bien LA valeur plantée qu'on retrouve ({epoque[n_pre, 0]})")
 809  
 810          # Le même test, décalé d'une demi-période d'échantillonnage : un marqueur ne tombe jamais
 811          # pile sur un échantillon dans la vraie vie. On accepte 1 échantillon d'écart, pas plus.
 812          epoque = epoch_from_stream(eeg, ts, instant_du_pic + 0.002, fs,
 813                                    pre_s=P300_PRE_S, post_s=P300_EPOCH_S)
 814          position = int(np.argmax(epoque[:, 0]))
 815          chk(abs(position - n_pre) <= 1,
 816              f"un marqueur entre deux échantillons reste aligné à ±1 ({position} vs {n_pre})")
 817  
 818          # --- LE MÊME test, mais par le VRAI chemin d'appel (_encaisser_flash, pas un appel direct
 819          # à epoch_from_stream) --------------------------------------------------------------------
 820          # Le test ci-dessus prouve que `epoch_from_stream` positionne juste QUAND on lui donne les
 821          # bons pre_s/post_s — il ne prouve PAS que `_encaisser_flash` les lui TRANSMET dans le bon
 822          # ordre. Piège concret, pas théorique : P300_PRE_S (0,15 s -> 38 éch.) et P300_EPOCH_S
 823          # (0,80 s -> 200 éch.) ont des tailles différentes, mais 38+200 == 200+38 == 238 — une
 824          # INVERSION `pre_s=self.post_s, post_s=self.pre_s` à l'appel réel produirait une époque de
 825          # la MÊME FORME (238, 8), invisible à tout contrôle de taille ou de bornes. Seule la
 826          # POSITION du signal DANS l'époque que le runtime a RÉELLEMENT construite peut l'attraper —
 827          # d'où ce second test, qui repasse par `rt.tick()` (donc par `_encaisser_flash`) au lieu
 828          # d'appeler le décodeur en direct comme ci-dessus.
 829          rt._vider_manche()
 830          moteur_aligne = _FauxMoteur(eeg, ts)
 831          moteur_aligne._lots = [[marqueur(instant_du_pic, 0)]]
 832          rt.tick(moteur_aligne, lsl_ts=instant_du_pic, now=999.0)
 833          chk(len(rt._epoques) == 1,
 834              f"le flash a produit UNE époque en passant par _encaisser_flash, le vrai chemin "
 835              f"d'appel du runtime ({len(rt._epoques)})")
 836          position_reelle = int(np.argmax(rt._epoques[-1][:, 0])) if rt._epoques else -1
 837          chk(position_reelle == n_pre,
 838              f"⚠️ ALIGNEMENT (chemin réel _encaisser_flash) : le pic se retrouve à l'échantillon "
 839              f"{position_reelle}, il devait être à {n_pre} (décalage de "
 840              f"{position_reelle - n_pre} échantillons = "
 841              f"{(position_reelle - n_pre) / fs * 1000:+.0f} ms) — pre_s/post_s mal transmis par "
 842              f"le runtime")
 843          rt._vider_manche()
 844      finally:
 845          p300_models.modeles_disponibles = vrai_dispo
 846          shutil.rmtree(dossier, ignore_errors=True)
 847  
 848      print(f"[p300] VERDICT : {'OK' if ok else 'PROBLÈME'}")
 849      return ok
 850  
 851  
 852  if __name__ == "__main__":
 853      use_utf8_console()
 854      _sys.exit(0 if _selftest() else 1)
